# Credentials each declared source needs, as: <source>=<ENV_VAR>
#
# One line per source Badger claims to search. If a source is declared here its
# credential must exist, or the session refuses to start. The failure this
# prevents is silent: a source Badger cannot reach returns nothing, and
# "nothing found" is indistinguishable from "never looked".
#
# GitHub is reached through the declarative tools in tools/*.yaml, whose scripts
# open a Composio session. One key covers every Composio-backed source, so this
# line is what stands between a broken connection and a confidently empty
# answer. Comment it out only when the github_* tools are
# also removed from tools/.

github=COMPOSIO_API_KEY

# Vertex AI runs on Application Default Credentials rather than an API key, and
# gitagent has no pre-flight check for google-vertex — a partial setup fails
# late and unclearly. Project and location are the two that must be set.
model=GOOGLE_CLOUD_PROJECT

# gmail=COMPOSIO_API_KEY
# drive=COMPOSIO_API_KEY

# The Composio end-user whose connected accounts the tools use. Without it
# the tools fall back to a user with no connections, and every search returns
# empty rather than erroring — measured 2026-08-19 as an eval collapsing to
# 3/15 because .env lacked a trailing newline and this variable never loaded.
identity=BADGER_USER_ID
